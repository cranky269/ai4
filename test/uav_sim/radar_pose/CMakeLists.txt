cmake_minimum_required(VERSION 3.10)
project(pose_audit)
find_package(gazebo REQUIRED)
include_directories(${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(model_pose_audit SHARED model_pose_audit.cpp)
target_compile_features(model_pose_audit PRIVATE cxx_std_17)
target_link_libraries(model_pose_audit ${GAZEBO_LIBRARIES})
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs geometry_msgs)
include_directories(${catkin_INCLUDE_DIRS})
add_executable(invalid_pose_check invalid_pose_check.cpp)
target_compile_features(invalid_pose_check PRIVATE cxx_std_17)
target_link_libraries(invalid_pose_check ${GAZEBO_LIBRARIES} ${catkin_LIBRARIES})
